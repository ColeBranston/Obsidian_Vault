-- SELECT and basic operators
SELECT col1, col2, SUM(col3) AS total
FROM table_name
WHERE col4 = 'value'
  AND col5 > 100
GROUP BY col1, col2
HAVING COUNT(*) > 1
ORDER BY total DESC
LIMIT 10;

-- JOIN variants
FROM tableA a
JOIN tableB b ON a.id = b.a_id            -- INNER JOIN
LEFT JOIN tableB b ON a.id = b.a_id       -- LEFT OUTER JOIN
RIGHT JOIN tableB b ON a.id = b.a_id      -- RIGHT OUTER JOIN
FULL JOIN tableB b ON a.id = b.a_id       -- FULL OUTER JOIN
CROSS JOIN tableB b                       -- Cartesian product

-- Subqueries
col IN (SELECT col FROM other_table WHERE condition)

-- EXISTS vs IN
WHERE EXISTS (
  SELECT 1
  FROM other_table
  WHERE other_table.fk_id = main_table.id
)

-- CASE expression
CASE
  WHEN condition1 THEN result1
  WHEN condition2 THEN result2
  ELSE default
END

-- Window functions
ROW_NUMBER() OVER (PARTITION BY colX ORDER BY colY DESC)
RANK(), DENSE_RANK(), NTILE(n)
LAG(col) OVER (PARTITION BY X ORDER BY Y)
LEAD(col) OVER (PARTITION BY X ORDER BY Y)
SUM(col)  OVER (PARTITION BY X ORDER BY Y ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)

-- String functions (PostgreSQL / MySQL variants)
CONCAT(col1, ' ', col2)
SUBSTRING(col FROM pos FOR len)   -- or SUBSTR(col, pos, len)
LEFT(col, n), RIGHT(col, n)
LOWER(col), UPPER(col), TRIM(col)
COALESCE(col, default), NULLIF(a, b)

-- Date functions
CURRENT_DATE, CURRENT_TIMESTAMP
DATE_TRUNC('month', timestamp_col)
EXTRACT(YEAR FROM date_col)
DATE_PART('dow', date_col)
DATEDIFF(day, date1, date2)      -- SQL Server, MySQL: DATEDIFF(date2,date1)
DATE_ADD(date_col, INTERVAL 3 DAY)
