
Below is a concise example of a typical ETL-style PySpark job you might describe in an interview:

```Python
from pyspark.sql import SparkSession
from pyspark.sql.functions import col, when, lit

def main():
    spark = SparkSession.builder \
        .appName("CustomerTransactionsETL") \
        .master("local[*]") \
        .config("spark.sql.shuffle.partitions", 50) \
        .getOrCreate()

    # 1. Read raw transaction data from Parquet
    transactions = spark.read.parquet("s3a://data-bucket/transactions/")

    # 2. Filter out invalid or fraudulent transactions
    valid_tx = transactions.filter(
        (col("amount") > 0) &
        (col("status") == "COMPLETED")
    )

    # 3. Enrich with user data (dimension table)
    users = spark.read.jdbc(
        url="jdbc:postgresql://db.example.com:5432/customers",
        table="public.users",
        properties={"user": "readonly", "password": "*****"}
    )

    joined = valid_tx.join(
        users, valid_tx.user_id == users.id, "inner"
    ).select(
        valid_tx.tx_id,
        valid_tx.user_id,
        users.country,
        valid_tx.amount,
        valid_tx.timestamp
    )

    # 4. Derive new columns (e.g., categorize high/low spender)
    enriched = joined.withColumn(
        "spender_category",
        when(col("amount") > 1000, lit("High"))
         .when(col("amount") > 100, lit("Medium"))
         .otherwise(lit("Low"))
    )

    # 5. Aggregate: total and average transaction by country
    agg_by_country = enriched.groupBy("country") \
        .agg(
            {"amount": "sum", "*": "count"}
        ) \
        .withColumnRenamed("sum(amount)", "total_amount") \
        .withColumnRenamed("count(1)", "num_transactions")

    # 6. Cache intermediate result if reused
    agg_by_country.cache()

    # 7. Write results to Parquet in date‐partitioned folders
    agg_by_country \
        .withColumn("date", col("timestamp").cast("date")) \
        .write \
        .mode("overwrite") \
        .partitionBy("date") \
        .parquet("s3a://analytics‐bucket/tx_summary/")

    # 8. Also register as a temp view for ad-hoc queries
    agg_by_country.createOrReplaceTempView("country_tx_summary")
    spark.sql("SELECT * FROM country_tx_summary WHERE total_amount > 1000000").show()

    # 9. Stop Spark
    spark.stop()

if __name__ == "__main__":
    main()

```

**Points to highlight:**

- Reading from Parquet (columnar, compressed) vs CSV.
- Filtering and joining with a dimension table (JDBC source).
- Deriving new columns with `when` expressions (built-in functions).
- Using `groupBy` + `agg` instead of writing custom reduceByKey.
- Caching `agg_by_country` if we query it more than once.
- Writing out partitioned Parquet for efficient downstream reads.
- Creating a temp view to mix SQL and DataFrame APIs.
